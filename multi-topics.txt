# ----------------------------------------------------------
# 1) BUILT-IN MATH FUNCTIONS
# ----------------------------------------------------------
# Python provides many built-in functions for working with numbers

x = 5
y = -5
z = 10

# round() → rounds a float to nearest integer
print(round(20.6))          # 21

# abs() → converts negative to positive
print(abs(y))              # 5

# pow(base, power) → base raised to power
print(pow(x, 2))           # 25
print(pow(4, 3))           # 64

# min() & max() → get smallest and largest number
print(min(3, 6, 1))        # 1
print(max(3, 6, 1))        # 6


# ----------------------------------------------------------
# 2) LOGICAL OPERATORS
# ----------------------------------------------------------
# Used to check multiple conditions
# or  → at least one condition must be True
# and → all conditions must be True
# not → reverses True to False and False to True

temp = 36
is_raining = False

# Using OR
if temp > 35 or temp < 0 or is_raining:
    print("Outdoor event is cancelled")
else:
    print("Outdoor event is scheduled")

# Using AND
temp = 30
is_sunny = True

if temp >= 30 and is_sunny:
    print("It is HOT and SUNNY")
else:
    print("Weather is normal")

# Using NOT
is_cloudy = True

if not is_cloudy:
    print("Sky is clear")
else:
    print("Sky is cloudy")

# 11) MINI PROJECT – CALCULATOR
# ----------------------------------------------------------

operator = input("Enter operator (+, -, *, /): ")
num_1 = float(input("Enter first number: "))
num_2 = float(input("Enter second number: "))

# Prevent division by zero
if operator == "/" and num_2 == 0:
    print("Cannot divide by zero")
    exit()

# Perform operation
if operator == "+":
    result = num_1 + num_2
elif operator == "-":
    result = num_1 - num_2
elif operator == "*":
    result = num_1 * num_2
elif operator == "/":
    result = num_1 / num_2
else:
    print("Invalid operator")
    exit()

# round() used to limit decimal places
print(f"Result = {round(result, 3)}")
